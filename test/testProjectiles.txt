\tpt cabin basement
\set self hp 10000

\\ Make probabilistic projectile branches deterministic for this script
\exe game.originalPercentChance=Core.percentChance
\exe game.originalRoll=Core.roll
\exe Core.percentChance=lambda chance:chance>=50
\exe Core.roll=lambda sides,modifier=0:1+modifier

\\ Lob an inventory Item at an object target
\spn Core.Creature("dummy","A patient target dummy with unusually deep reserves.",100,[1]*10,hp=10000,status=[["tamed",-1]])
\spn Core.Item("dart","A dart eager to become somebody else's problem.",4,-1,"iron")
take dart
\lob dart dummy
\chk fo("dart",queryType="room").parent is player.parent
\chk fo("dart",queryType="room") not in player
\chk fo("dummy",queryType="room").hp < 10000

\\ Lob a carried Creature and clear every carrying slot
\spn Core.Creature("captive","A compact captive bracing for unconventional travel.",4,[1]*10,status=[["tamed",-1]])
take captive
\chk fo("captive",queryType="room").carrier is player
\lob captive dummy
\chk fo("captive",queryType="room").carrier is None
\chk fo("captive",queryType="room") not in player.carrying()
\chk all(player[slot] is not fo("captive",queryType="room") for slot in player.carrySlots)

\\ Lob a riding Creature and clear its riding tether
\spn Core.Creature("pony","A pony serving briefly as launch infrastructure.",80,[10]*10,status=[["tamed",-1]])
\spn Core.Creature("jockey","A jockey discovering an obscure clause in the job description.",4,[1]*10,status=[["tamed",-1]])
take jockey
put jockey on pony
\chk fo("jockey",queryType="room").riding is fo("pony",queryType="room")
\lob jockey dummy
\chk fo("jockey",queryType="room").riding is None
\chk fo("jockey",queryType="room").parent is player.parent

\\ Homing and calamity can ricochet back to the launcher, who catches it
\spn Core.Creature("homingdummy","A nimble dummy that sends enchanted missiles home.",30,[1000]*10,hp=10000,status=[["tamed",-1]])
\spn Core.Item("seeker","A missile enchanted to miss in a highly intentional manner.",4,-1,"iron",status=[["homing",-1]])
\mbu self calamity -1
take seeker
throw seeker at homingdummy
yes
\chk fo("seeker",queryType="player").parent is player
\chk fo("seeker",queryType="player") in player.gear.values()
\xpl calamity

\\ A forced miss has a valid launcher and can ricochet into a selected victim
\tpt cabin basement
\spn Core.Creature("missdummy","A dummy positioned to be missed with confidence.",30,[1000]*10,hp=10000,status=[["tamed",-1]])
\spn Core.Creature("victim","A volunteer wearing the curse of narrative inevitability.",30,[1]*10,hp=10000,status=[["tamed",-1],["calamity",-1]])
\spn Core.Item("boomerang","A boomerang with only a theoretical understanding of returning.",4,-1,"wood")
take boomerang
throw boomerang at missdummy
\chk fo("victim",queryType="room").hp < 10000
\chk fo("boomerang",queryType="room").parent is player.parent
