\\ Failed ordinary movement leaves the object and destination unchanged
\spn Items.Box("packed trunk","A trunk packed according to an aggressively literal inventory.",10,-1,"wood",[],capacity=5)
\spn Core.Item("resident brick","A brick with permanent residency paperwork.",5,-1,"stone")
take resident brick
put resident brick in packed trunk
\spn Core.Item("hopeful parcel","A parcel optimistic about limited storage.",4,-1,"cloth")
\exe game.ordinaryMoveSucceeded=fo("hopeful parcel",queryType="room").changeLocation(fo("packed trunk",queryType="room"))
\chk not game.ordinaryMoveSucceeded
\chk fo("hopeful parcel",queryType="room").parent is player.parent
\chk fo("packed trunk",queryType="room").items == [fo("resident brick",queryType="room",roomD=3)]

\\ Invalid direct add invokes displacement but preserves an outward destination
\exe game.invalidAddSucceeded=fo("packed trunk",queryType="room").add(fo("hopeful parcel",queryType="room"))
\chk not game.invalidAddSucceeded
\chk fo("hopeful parcel",queryType="room").parent is player.parent
\chk fo("hopeful parcel",queryType="room") in player.parent.items
\chk fo("hopeful parcel",queryType="room").id in game.itemRegistry

\\ Displacement searches outward through nested containers
\spn Items.Box("outer hamper","A hamper that has already given all available room to another box.",15,-1,"wicker",[],capacity=15)
\spn Items.Box("inner hamper","A hamper nesting where no hamper should.",15,-1,"wicker",[],capacity=5)
\spn Core.Item("expanding accordion","An accordion taking a deep and spatially inconvenient breath.",3,-1,"wood")
take inner hamper
put inner hamper in outer hamper
\exe game.accordion=fo("expanding accordion",queryType="room")
\exe game.innerHamper=fo("inner hamper",queryType="room",roomD=3)
\exe player.parent.remove(game.accordion)
\exe game.innerHamper.add(game.accordion)
\grw expanding accordion 4
\exe game.accordion.displace(game.innerHamper.parent)
\chk game.accordion.parent is player.parent
\chk game.accordion.platform is player.parent.floor
\chk game.innerHamper.parent is fo("outer hamper",queryType="room")
