\\ Positive status durations decrement and expire; permanent statuses remain
\mbu self fleeting 2
\mbu self enduring -1
\chk player.hasStatus("fleeting",2)
\exe game.passTime(1)
\chk player.hasStatus("fleeting",1)
\chk player.hasStatus("enduring",-1)
\exe game.passTime(1)
\chk not player.hasStatus("fleeting")
\chk player.hasStatus("enduring",-1)
\xpl enduring

\\ Nonstacking statuses refuse duplicates and exact-duration removal preserves siblings
\exe game.firstSingleClockAdded=player.addStatus("single clock",5,stackable=False,silent=True)
\exe game.secondSingleClockAdded=player.addStatus("single clock",10,stackable=False,silent=True)
\chk game.firstSingleClockAdded
\chk not game.secondSingleClockAdded
\chk player.hasStatus("single clock",5)
\exe player.addStatus("split clock",5,silent=True)
\exe player.addStatus("split clock",10,silent=True)
\exe player.removeStatus("split clock",5,silent=True)
\chk not player.hasStatus("split clock",5)
\chk player.hasStatus("split clock",10)
\exe player.removeStatus("single clock",silent=True)
\exe player.removeStatus("split clock",silent=True)

\\ Damaging conditions apply damage while active and stop after removal
\set self hp 10000
\mbu self poisoned -1
\exe game.beforePoison=player.hp
\exe player.passTime(1)
\chk player.hp < game.beforePoison
\xpl poisoned
\exe game.afterPoison=player.hp
\exe player.passTime(1)
\chk player.hp >= game.afterPoison

\\ Hunger advances to starvation and clears after eating time is reset
\exe game.time=player.lastAte+51+5*player.ENDR()
\exe player.checkHungry()
\chk player.hasStatus("hungry",-2)
\exe game.time=player.lastAte+101+10*player.ENDR()
\exe player.checkHungry()
\chk player.hasStatus("starving",-2)
\chk not player.hasStatus("hungry",-2)
\exe player.lastAte=game.time
\exe player.checkHungry()
\chk not player.hasAnyStatus("hungry","starving")

\\ Breath holding advances to breathless and drowning, then recovers
\exe player.lastBreathed=0
\exe game.time=player.ENDR()//3+1
\exe player.lastAte=game.time
\exe player.lastSlept=game.time
\exe player.checkBreathing()
\chk player.hasStatus("breathless",-2)
\exe game.time=player.ENDR()//2+1
\exe player.checkBreathing()
\chk player.hasStatus("drowning",-2)
\chk not player.hasStatus("breathless",-2)
\exe player.lastBreathed=game.time
\exe player.checkBreathing()
\chk not player.hasAnyStatus("breathless","drowning")
